const express = require('express'); const cors = require('cors'); const rateLimit = require('express-rate-limit'); const app = express(); app.use(express.json()); app.use(cors()); const limiter = rateLimit({ windowMs: 60 * 1000, max: 30, message: { error: 'Troppi calcoli, aspetta un minuto.' } }); app.use('/calculate', limiter); app.post('/calculate', async (req, res) => { const { query, history } = req.body; if (!query || typeof query !== 'string' || query.length > 500) { return res.status(400).json({ error: 'Input non valido.' }); } const messages = [ { role: 'system', content: `Sei una calcolatrice AI con memoria della conversazione. Tieni traccia dei risultati precedenti e usali come base per i calcoli successivi. Il contesto include [Risultato precedente: X] prima della query — usalo come punto di partenza quando la query e relativa (es. "meno 8", "dividi per 2"). L'input puo arrivare da riconoscimento vocale: converti parole in numeri (es. "due virgola cinquanta" = 2.50, "venti" = 20). Regole decimali: mostra sempre fino a 6 cifre decimali significative (es. 0.333333). Se l'utente specifica nella query un numero diverso di decimali (es. "con 2 decimali", "arrotonda a 3", "senza decimali"), rispetta quella richiesta SOLO per questa singola risposta, poi torna a 6 decimali. Rispondi SEMPRE e SOLO con JSON valido, nessun testo fuori, nessun markdown: {"result":,"explanation":""} Se non e un calcolo: {"result":"?","explanation":"Non e un calcolo"}` } ]; if (Array.isArray(history) && history.length > 0) { history.slice(-10).forEach(h => { messages.push({ role: 'user', content: h.q }); messages.push({ role: 'assistant', content: `Il risultato e ${h.a}. ${h.explanation || ''}` }); }); } const lastResult = Array.isArray(history) && history.length > 0 ? history[0].a : null; const contextQuery = lastResult !== null ? `[Risultato precedente: ${lastResult}] ${query}` : query; messages.push({ role: 'user', content: contextQuery }); try { const response = await fetch('https://api.groq.com/openai/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${process.env.GROQ_API_KEY}` }, body: JSON.stringify({ model: 'llama-3.3-70b-versatile', temperature: 0, max_tokens: 200, messages }) }); const data = await response.json(); if (data.error) return res.status(500).json({ error: data.error.message }); const text = data.choices[0].message.content; const parsed = JSON.parse(text.replace(/```json|```/g, '').trim()); res.json(parsed); } catch (e) { res.status(500).json({ error: 'Errore interno: ' + e.message }); } }); app.get('/health', (_, res) => res.json({ status: 'ok' })); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`CalcAI backend running on port ${PORT}`));
